const xorEncrypt = (plainText) => {
    const key = getOrCreateCipherKey();
    const chars = Array.from(plainText).map((ch, i) =>
        String.fromCharCode(ch.charCodeAt(0) ^ key.charCodeAt(i % key.length))
    );
    return `enc:${btoa(chars.join(""))}`;
};

const xorDecrypt = (cipherText) => {
    if (!String(cipherText || "").startsWith("enc:")) return cipherText;
    const raw = atob(String(cipherText).slice(4));
    const decode = (key) =>
        Array.from(raw)
            .map((ch, i) => String.fromCharCode(ch.charCodeAt(0) ^ key.charCodeAt(i % key.length)))
            .join("");
    try {
        return decode(getOrCreateCipherKey());
    } catch {
        try {
            const legacy = localStorage.getItem(OFFLINE_CIPHER_KEY);
            if (legacy) return decode(legacy);
        } catch {
            // no-op
        }
        return "[]";
    }
};

const hashString = (value = "") => {
    let hash = 2166136261;
    const input = String(value || "");
    for (let i = 0; i < input.length; i += 1) {
        hash ^= input.charCodeAt(i);
        hash += (hash << 1) + (hash << 4) + (hash << 7) + (hash << 8) + (hash << 24);
    }
    return Math.abs(hash >>> 0).toString(36);
};

const replaceTempIds = (value, idMap) => {
    if (!value) return value;
    if (typeof value === "string") {
        return idMap[value] || value;
    }
    if (Array.isArray(value)) {
        return value.map((entry) => replaceTempIds(entry, idMap));
    }
    if (typeof value === "object") {
        const next = {};
        Object.entries(value).forEach(([key, val]) => {
            const replaced = replaceTempIds(val, idMap);
            next[key] = replaced;
        });
        return next;
    }
    return value;
};

const replaceTempIdsInUrl = (value, idMap) => {
    if (!value || typeof value !== "string") return value;
    let next = value;
    Object.entries(idMap || {}).forEach(([tempId, realId]) => {
        if (!tempId || !realId) return;
        next = next.split(tempId).join(String(realId));
    });
    return next;
};

const classifyDeadLetterReason = ({ status, data, fallbackMessage, item }) => {
    const code = String(data?.code || data?.type || "").toUpperCase();
    const message = String(data?.message || data?.error || fallbackMessage || "").toLowerCase();
    const path = String(item?.url || "").toLowerCase();

    if (
        code.includes("INSUFFICIENT_STOCK") ||
        message.includes("insufficient stock") ||
        message.includes("not enough stock")
    ) {
        return {
            reasonCode: "INSUFFICIENT_STOCK",
            resolutionHint: "Reduce quantities or re-stock before retrying.",
        };
    }

    if (
        code.includes("PRICE_CHANGED") ||
        message.includes("price changed") ||
        message.includes("price mismatch")
    ) {
        return {
            reasonCode: "PRICE_CHANGED",
            resolutionHint: "Refresh product prices and update order payload before retrying.",
        };
    }

    if (
        code.includes("CUSTOMER_NOT_FOUND") ||
        message.includes("customer not found") ||
        message.includes("unknown customer")
    ) {
        return {
            reasonCode: "CUSTOMER_MISSING",
            resolutionHint: "Set customer to walk-in or select a valid customer.",
        };
    }

    if (status === 413 || message.includes("too large") || message.includes("payload too large")) {
        return {
            reasonCode: "REQUEST_TOO_LARGE",
            resolutionHint: "Split the operation into smaller batches and retry.",
        };
    }

    if (path.includes("/orders")) {
        return {
            reasonCode: "ORDER_SYNC_FAILED",
            resolutionHint: "Review order payload and retry.",
        };
    }

    return {
        reasonCode: "UNKNOWN_SYNC_ERROR",
        resolutionHint: "Review payload and retry from queue.",
    };
};

const migrateOfflineStorageIfNeeded = () => {

        if (version >= OFFLINE_QUEUE_VERSION) return;

        const decodeLegacy = (raw, fallback = []) => {
            if (!raw) return fallback;
            if (String(raw).startsWith("enc:")) {
                try {
                    return JSON.parse(xorDecrypt(raw) || "[]");
                } catch {
                    return fallback;
                }
            }
            try {
                return JSON.parse(raw);
            } catch {
                return fallback;
            }
        };

        const legacyQueueRaw = localStorage.getItem(OFFLINE_QUEUE_KEY);
        const legacyDeadRaw = localStorage.getItem(OFFLINE_DEAD_LETTER_KEY);
        const legacyResultsRaw = localStorage.getItem(OFFLINE_SYNC_RESULTS_KEY);

        const plainQueue = decodeLegacy(legacyQueueRaw, []);
        const plainDead = decodeLegacy(legacyDeadRaw, []);
        const plainResults = decodeLegacy(legacyResultsRaw, []);

        if (legacyQueueRaw && !localStorage.getItem(resolveOfflineKey(OFFLINE_QUEUE_KEY))) {
            localStorage.setItem(resolveOfflineKey(OFFLINE_QUEUE_KEY), xorEncrypt(JSON.stringify(plainQueue)));
        }
        if (legacyDeadRaw && !localStorage.getItem(resolveOfflineKey(OFFLINE_DEAD_LETTER_KEY))) {
            localStorage.setItem(resolveOfflineKey(OFFLINE_DEAD_LETTER_KEY), xorEncrypt(JSON.stringify(plainDead)));
        }
        if (legacyResultsRaw && !localStorage.getItem(resolveOfflineKey(OFFLINE_SYNC_RESULTS_KEY))) {
            localStorage.setItem(resolveOfflineKey(OFFLINE_SYNC_RESULTS_KEY), xorEncrypt(JSON.stringify(plainResults)));
        }
        localStorage.setItem(resolveOfflineKey(OFFLINE_QUEUE_VERSION_KEY), String(OFFLINE_QUEUE_VERSION));
    } catch {
        // no-op
    }
};

const queueOfflineMutation = (config) => {
    const queue = loadOfflineQueue();
    const existingIdempotency = config.headers?.["X-Idempotency-Key"] || config.headers?.["x-idempotency-key"];
    const idempotencyKey = existingIdempotency || generateIdempotencyKey();
    const clientTempId = config.__clientTempId || config.headers?.["X-Client-Temp-Id"];
    const item = {

        params: config.params ?? null,

        },

        nextRetryAt: 0,
        status: "queued",
    };

        method: String(item.method || "").toUpperCase(),

        message: "Queued while offline",
    });

const flushOfflineQueue = async () => {
    if (isFlushingOfflineQueue) return;
    if (!effectiveOnline()) return;
    const queue = loadOfflineQueue();
    if (!queue.length) return;
    isFlushingOfflineQueue = true;
    setSyncState({
        inProgress: true,
        startedAt: nowTs(),
        processedCount: 0,
        totalToProcess: queue.length,
        currentItemId: null,
    });
    const remaining = [];
    const deadLetter = loadDeadLetterQueue();
    const currentTime = nowTs();
    const idMap = loadOfflineIdMap();
    let processedCount = 0;
    for (const item of queue) {
        if (Number(item.nextRetryAt || 0) > currentTime) {
            remaining.push(item);
            continue;
        }
        setSyncState({
            inProgress: true,
            startedAt: nowTs(),
            processedCount,
            totalToProcess: queue.length,
            currentItemId: item.id,
        });
        try {
            const preparedData = replaceTempIds(item.data, idMap);
            const preparedParams = replaceTempIds(item.params, idMap);
            const preparedUrl = replaceTempIdsInUrl(item.url, idMap);
            const response = await axiosConfig.request({
                method: item.method,
                url: preparedUrl,
                data: preparedData,
                params: preparedParams,
                headers: item.headers,
                __skipOfflineQueue: true,
                __fromOfflineQueue: true,
            });
            const resolvedId = extractResponseId(response?.data);
            if (item.clientTempId && resolvedId) {
                registerOfflineIdMapping(item.clientTempId, resolvedId);
                idMap[item.clientTempId] = resolvedId;
            }
            markOfflineAuditSynced(item.id);
            appendSyncResult({
                queueItemId: item.id,
                url: item.url,
                method: String(item.method || "").toUpperCase(),
                status: "SYNCED",
                message: "Successfully synced",
            });
        } catch (err) {
            const status = err?.response?.status;
            const data = err?.response?.data;
            if (isDuplicateConflict(status, data)) {
                const resolvedId = extractResponseId(data);
                if (item.clientTempId && resolvedId) {
                    registerOfflineIdMapping(item.clientTempId, resolvedId);
                    idMap[item.clientTempId] = resolvedId;
                }
                appendSyncResult({
                    queueItemId: item.id,
                    url: item.url,
                    method: String(item.method || "").toUpperCase(),
                    status: "DUPLICATE",
                    message: "Skipped duplicate on sync",
                });
                continue;
            }
            if (!err?.response) {
                // Still offline/network unstable; keep current + remaining queue.
                remaining.push(item, ...queue.slice(queue.indexOf(item) + 1));
                break;
            }
            const nextRetries = Number(item.retries || 0) + 1;
            const failMessage = data?.message || data?.error || `HTTP ${status}`;
            if (nextRetries >= OFFLINE_MAX_RETRIES) {
                const classified = classifyDeadLetterReason({
                    status,
                    data,
                    fallbackMessage: failMessage,
                    item,
                });
                deadLetter.push({
                    ...item,
                    retries: nextRetries,
                    lastError: failMessage,
                    movedToDeadLetterAt: nowTs(),
                    status: "dead_letter",
                    reasonCode: classified.reasonCode,
                    resolutionHint: classified.resolutionHint,
                });
                appendSyncResult({
                    queueItemId: item.id,
                    url: item.url,
                    method: String(item.method || "").toUpperCase(),
                    status: "DEAD_LETTER",
                    message: `${classified.reasonCode}: ${failMessage}`,
                });
                sendSyncAlertEvent({
                    type: "SYNC_CONFLICT",
                    message: classified.reasonCode,
                    details: failMessage,
                    queueDepth: queue.length,
                    deadLetterCount: deadLetter.length,
                    failureCount: nextRetries,
                });
            } else {
                remaining.push({
                ...item,
                retries: nextRetries,
                status: "retry_pending",
                nextRetryAt: nowTs() + computeRetryDelayMs(nextRetries),
                lastError: failMessage,
            });
                appendSyncResult({
                    queueItemId: item.id,
                    url: item.url,
                    method: String(item.method || "").toUpperCase(),
                    status: "FAILED",
                    message: failMessage,
                });
                if (nextRetries === 3) {
                    sendSyncAlertEvent({
                        type: "QUEUE_FAILURE",
                        message: "Queue item failed multiple times",
                        details: failMessage,
                        queueDepth: queue.length,
                        deadLetterCount: deadLetter.length,
                        failureCount: nextRetries,
                    });
                }
            }
        }
        processedCount += 1;
    }

    saveOfflineQueue(remaining);
    saveDeadLetterQueue(deadLetter);
    if (remaining.length === 0) {
        saveLastOfflineSyncTs(Date.now());
    }
    isFlushingOfflineQueue = false;
    setSyncState({
        inProgress: false,
        endedAt: nowTs(),
        processedCount,
        totalToProcess: queue.length,
        currentItemId: null,
    });
};

const evaluateQueueAlertConditions = () => {
    const settings = readAlertSettingsCache();
    const deviceOverride = readAlertDeviceOverride();
    const effectiveSettings = deviceOverride?.enabled
        ? {
            ...settings,
            deadLetterThreshold: deviceOverride.deadLetterThreshold ?? settings.deadLetterThreshold,
            queueStuckMinutes: deviceOverride.queueStuckMinutes ?? settings.queueStuckMinutes,
        }
        : settings;
    const queue = loadOfflineQueue();
    const deadLetter = loadDeadLetterQueue();
    const sync = readSyncState();
    const lastSyncAt = readLastOfflineSyncTs();

    if (effectiveSettings.deadLetterThresholdEnabled && Number(effectiveSettings.deadLetterThreshold) > 0) {
        if (deadLetter.length >= Number(effectiveSettings.deadLetterThreshold)) {
            sendSyncAlertEvent({
                type: "DEAD_LETTER_THRESHOLD",
                message: "Dead-letter queue exceeded threshold",
                details: `Dead-letter size ${deadLetter.length} reached threshold ${effectiveSettings.deadLetterThreshold}`,
                queueDepth: queue.length,
                deadLetterCount: deadLetter.length,
                failureCount: 0,
            });
        }
    }

    if (effectiveSettings.queueStuckEnabled && Number(effectiveSettings.queueStuckMinutes) > 0) {
        const minutes = Number(effectiveSettings.queueStuckMinutes);
        const oldestQueueTs = queue.length
            ? Math.min(...queue.map((item) => Number(item.createdAt || Date.now())))
            : 0;
        const baselineTs = lastSyncAt || oldestQueueTs || 0;
        const ageMs = baselineTs ? Date.now() - baselineTs : 0;
        if (
            queue.length > 0 &&
            !sync?.inProgress &&
            baselineTs &&
            ageMs > minutes * 60 * 1000
        ) {
            sendSyncAlertEvent({
                type: "QUEUE_STUCK",
                message: "Offline queue stuck",
                details: `Queue has ${queue.length} item(s) and last sync was ${Math.round(ageMs / 60000)} minutes ago.`,
                queueDepth: queue.length,
                deadLetterCount: deadLetter.length,
                failureCount: 0,
            });
        }
    }
};

export const retryOfflineQueueItem = async (id) => {
    const queue = loadOfflineQueue();
    const index = queue.findIndex((item) => item.id === id);
    if (index === -1) return { ok: false, reason: "NOT_FOUND" };
    const item = queue[index];
    if (!effectiveOnline()) return { ok: false, reason: "OFFLINE" };

    try {
        const idMap = loadOfflineIdMap();
        const preparedData = replaceTempIds(item.data, idMap);
        const preparedParams = replaceTempIds(item.params, idMap);
        const preparedUrl = replaceTempIdsInUrl(item.url, idMap);
        await axiosConfig.request({
            method: item.method,
            url: preparedUrl,
            data: preparedData,
            params: preparedParams,
            headers: item.headers,
            __skipOfflineQueue: true,
            __fromOfflineQueue: true,
        });
        markOfflineAuditSynced(item.id);
        queue.splice(index, 1);
        saveOfflineQueue(queue);
        if (queue.length === 0) {
            saveLastOfflineSyncTs(Date.now());
        }
        appendSyncResult({
            queueItemId: item.id,
            url: item.url,
            method: String(item.method || "").toUpperCase(),
            status: "SYNCED",
            message: "Manual retry success",
        });
        return { ok: true };
    } catch (err) {
        const status = err?.response?.status;
        const data = err?.response?.data;
        if (isDuplicateConflict(status, data)) {
            const resolvedId = extractResponseId(data);
            if (item.clientTempId && resolvedId) {
                registerOfflineIdMapping(item.clientTempId, resolvedId);
            }
            queue.splice(index, 1);
            saveOfflineQueue(queue);
            appendSyncResult({
                queueItemId: item.id,
                url: item.url,
                method: String(item.method || "").toUpperCase(),
                status: "DUPLICATE",
                message: "Manual retry duplicate",
            });
            return { ok: true, duplicate: true };
        }
        const nextRetries = Number(item.retries || 0) + 1;
        const failMessage = data?.message || data?.error || (status ? `HTTP ${status}` : "NETWORK_ERROR");
        if (nextRetries >= OFFLINE_MAX_RETRIES) {
            const classified = classifyDeadLetterReason({
                status,
                data,
                fallbackMessage: failMessage,
                item,
            });
            const dead = loadDeadLetterQueue();
            dead.push({
                ...item,
                retries: nextRetries,
                lastError: failMessage,
                movedToDeadLetterAt: nowTs(),
                status: "dead_letter",
                reasonCode: classified.reasonCode,
                resolutionHint: classified.resolutionHint,
            });
            queue.splice(index, 1);
            saveDeadLetterQueue(dead);
            saveOfflineQueue(queue);
            appendSyncResult({
                queueItemId: item.id,
                url: item.url,
                method: String(item.method || "").toUpperCase(),
                status: "DEAD_LETTER",
                message: `${classified.reasonCode}: ${failMessage}`,
            });
            sendSyncAlertEvent({
                type: "SYNC_CONFLICT",
                message: classified.reasonCode,
                details: failMessage,
                queueDepth: queue.length,
                deadLetterCount: dead.length,
                failureCount: nextRetries,
            });
            return { ok: false, reason: "DEAD_LETTER" };
        }
        queue[index] = {
            ...item,
            retries: nextRetries,
            status: "retry_pending",
            nextRetryAt: nowTs() + computeRetryDelayMs(nextRetries),
            lastError: failMessage,
        };
        saveOfflineQueue(queue);
        appendSyncResult({
            queueItemId: item.id,
            url: item.url,
            method: String(item.method || "").toUpperCase(),
            status: "FAILED",
            message: failMessage,
        });
        if (nextRetries === 3) {
            sendSyncAlertEvent({
                type: "QUEUE_FAILURE",
                message: "Queue item failed multiple times",
                details: failMessage,
                queueDepth: queue.length,
                deadLetterCount: loadDeadLetterQueue().length,
                failureCount: nextRetries,
            });
        }
        return { ok: false, reason: "FAILED" };
    }
};

    (config) => {
        if (isForcedOffline() && !config.__skipOfflineQueue) {
            return Promise.reject({
                config,
                code: "OFFLINE_TEST_MODE",
            });
        }
        const latency = getOfflineTestLatencyMs();
        if (latency > 0) {
            return new Promise((resolve) => {
                setTimeout(() => resolve(config), latency);
            });
        }
        const token = safeStorage.getItem(AUTH_TOKEN_KEY);
        if (token && !config.headers?.Authorization) {
            config.headers = config.headers || {};
            config.headers.Authorization = `Bearer ${token}`;
        }
        const method = String(config.method || "get").toLowerCase();
        if (isMutationMethod(method)) {
            config.headers = config.headers || {};
            if (!config.headers["X-Idempotency-Key"] && !config.headers["x-idempotency-key"]) {
                config.headers["X-Idempotency-Key"] = generateIdempotencyKey();
            }
        }
        return config;
    },

import { create } from "zustand";

/*
This is the core query store and this will be used by other query stores
i built it cause i did not know to use REACT QUERY..so wanted to understand it better

this will help in

-Scaling
-avoid duplicated data
-avoid race conditions by deduplicating requests per query storeKey
-response will have tokens so stall responses do not overwrite newone
-re using data accross screens
-caching
-errors
-loading states

first when i built it did not have selectors..and i was facing any infinty loop problem of re renders..
so i did some research and came up with a solution

//creating a store that accepts:
// extend - this will allow each feature to add its own extensions
// options - config for debug mode
export const createQueryStore = (extend, options = {}) =>
    create((set, get) => {

        //this is for turning debug mode on or off .
        //if the debug is true the store will log warnings for bad pattens
        const debug = !!(options && options.debug);

        //the following are closure variables

        //this variable remembers previos results arrays so that React does not re render
        const __listCache = {};

        //this one will track active api calls to avoid stale responses
        const __inflight = {};

        //this variable will be used in debugging by tracking how oftern arrays are recreated
        const __allocCounts = {};

        //this will delete illegal set() inside selectors to avoid re rendering infinty loops
        let __inSelector = false;

        /*
        This will wrap zustand set()
         -what i noticed was if you call set() inside a selector ,React will give an error of maximum update depth exceeded..loop
         -hence this code wont crash
         */
        const setTracked = (...args) => {
            return set(...args);
        };

        //initial store
        return {

            //used to store items by id
            entities: {},

            //used to map querykey to list of ids
            queries: {},

            //loading
            loading: {},

            //error
            error: {},

            //used for page, number page size,total pages
            pagination: {},

            //used to store total counts per query key
            totals: {},

            //CORE HELPERS

            /*
            this will mark a request as loading
            will clear old errors
            will create a unique token for the request
             */
            start: (key) => {
                const token = Symbol(key);
                __inflight[key] = token;
                setTracked((s) => ({
                    loading: { ...s.loading, [key]: true },
                    error: { ...s.error, [key]: null },
                }));
                return token;
            },

            /*
            this class is when request succeeds
            it ignores stale API responses
            normalizes data
            it then updates
             -entities
             -queries
             -loading
             */
            success: (key, items, token, meta = {}) =>
                setTracked((s) => {

                    // if a token was provided but it's not the current inflight token, ignore this stale response
                    if (token && __inflight[key] && __inflight[key] !== token) {
                        return s;
                    }

                    const safeItems = Array.isArray(items) ? items : [];

                    // will update the closure list cache for this storeKey to the new list reference
                    __listCache[key] = safeItems.map((item) => item);

                    // will clear inflight token for this storeKey since we have successfully applied the result
                    delete __inflight[key];

                    return {
                        //loading is set to false when succeded
                        loading: { ...s.loading, [key]: false },

                        //here we only store ids not objects
                        //it will prevent duplication
                        //enables reuse
                        queries: {
                            ...s.queries,
                            [key]: safeItems.map((item) => item.id),
                        },

                        entities: {
                            ...s.entities,
                            ...Object.fromEntries(
                                safeItems.map((item) => [item.id, item])
                            ),
                        },

                        totals: {
                            ...s.totals,
                            [key]: meta.total ?? safeItems.length,
                        },
                    };
                }),

            // fail also accepts token and only applies if token matches
            //stops loadinding
            //stores error
            // ignores stall response
            fail: (key, err, token) =>
                setTracked((s) => {
                    if (token && __inflight[key] && __inflight[key] !== token) {
                        return s;
                    }

                    delete __inflight[key];

                    return {
                        loading: { ...s.loading, [key]: false },
                        error: { ...s.error, [key]: err },
                    };
                }),

            //invalidation

            //clear evrything
            clear: () =>
                setTracked({
                    entities: {},
                    queries: {},
                    loading: {},
                    error: {},
                    pagination: {},
                    totals: {},
                }),

            //remove one query
            //used when data is outdated or refetching
            invalidate: (key) =>
                setTracked((s) => {

                    // clear closure caches for this key and paginated keys
                    Object.keys(__listCache).forEach((k) => {
                        if (k === key || k.startsWith(`${key}_page_`)) {
                            delete __listCache[k];
                        }
                    });

                    delete __inflight[key];

                    const { [key]: _, ...restQueries } = s.queries;
                    const { [key]: __, ...restLoading } = s.loading;
                    const { [key]: ___, ...restError } = s.error;
                    const { [key]: ____, ...restTotals } = s.totals;

                    return {
                        queries: restQueries,
                        loading: restLoading,
                        error: restError,
                        totals: restTotals,
                    };
                }),

            setEntity: (id, entity) =>
                setTracked((s) => ({
                    entities: { ...s.entities, [id]: entity },
                })),

            replaceId: (key, oldId, newId) =>
                setTracked((s) => ({
                    queries: {
                        ...s.queries,
                        [key]: s.queries[key].map((id) =>
                            id === oldId ? newId : id
                        ),
                    },
                })),

            setTotals: (key, total) =>
                setTracked((s) => ({
                    totals: { ...s.totals, [key]: total },
                })),


            //remove all queris starting with prefix
            //for bulk refresh
            invalidatePrefix: (prefix) =>
                setTracked((s) => {
                    const filter = (obj) =>
                        Object.fromEntries(
                            Object.entries(obj).filter(([k]) => !k.startsWith(prefix))
                        );

                    Object.keys(__listCache).forEach((k) => {
                        if (k.startsWith(prefix)) delete __listCache[k];
                    });
                    Object.keys(__inflight).forEach((k) => {
                        if (k.startsWith(prefix)) delete __inflight[k];
                    });

                    return {
                        queries: filter(s.queries),
                        loading: filter(s.loading),
                        error: filter(s.error),
                        totals: filter(s.totals),
                    };
                }),

            /*
            optimistically remove an entity from query list
             */
            optimisticRemove: (key, id) =>
                setTracked((s) => {
                    const ids = s.queries[key];
                    if (!ids) return s;

                    return {
                        queries: {
                            ...s.queries,
                            [key]: ids.filter((x) => x !== id),
                        },
                        totals: {
                            ...s.totals,
                            [key]: Math.max((s.totals[key] ?? ids.length) - 1, 0),
                        },
                    };
                }),

            optimisticMove(fromKey, toKey, entity) {
                setTracked((s) => {
                    const fromIds = s.queries[fromKey] ?? [];
                    const toIds = s.queries[toKey] ?? [];

                    return {
                        queries: {
                            ...s.queries,
                            [fromKey]: fromIds.filter((id) => id !== entity.id),
                            [toKey]: [entity.id, ...toIds],
                        },
                        totals: {
                            ...s.totals,
                            [fromKey]: Math.max(
                                (s.totals[fromKey] ?? fromIds.length) - 1,
                                0
                            ),
                            [toKey]: (s.totals[toKey] ?? toIds.length) + 1,
                        },
                        entities: {
                            ...s.entities,
                            [entity.id]: entity,
                        },
                    };
                });
            },

            /*
            rollback helper
             */
            restoreQuery: (key, prevIds) =>
                setTracked((s) => ({
                    queries: { ...s.queries, [key]: prevIds },
                    totals: {
                        ...s.totals,
                        [key]: prevIds?.length ?? 0,
                    },
                })),

            //Selector helpers

            /*
              -this will convert ids to full objects
              -prevents unnecessary re renders
              -it will also refuse same array refrence if data did not change
             */
            selectList:
                (key) =>
                    (s) => {
                        __inSelector = true;
                        try {
                            const ids = s.queries[key] || [];
                            const newList = ids.map((id) => s.entities[id]);

                            const cache = __listCache[key];
                            if (cache && cache.length === newList.length) {
                                let same = true;
                                for (let i = 0; i < newList.length; i++) {
                                    if (cache[i] !== newList[i]) {
                                        same = false;
                                        break;
                                    }
                                }
                                if (same) return cache;
                            }

                            __listCache[key] = newList;
                            return newList;
                        } finally {
                            __inSelector = false;
                        }
                    },

            //for loading returns true or false
            selectLoading: (key) => (s) => !!s.loading[key],

            //returns error messages or null
            selectError: (key) => (s) => s.error[key],

            selectPagination: (key) => (s) => s.pagination[key],
            selectById:
                (id) =>
                    (s) => s.entities[id] || null,

            selectPage:
                (key, page = 1, pageSize = 10) =>
                    (s) => {
                        const ids = s.queries[key] || [];
                        const start = (page - 1) * pageSize;
                        const end = start + pageSize;

                        const list = ids.slice(start, end).map((id) => s.entities[id]);

                        const cacheKey = `${key}_page_${page}_${pageSize}`;
                        const cache = __listCache[cacheKey];

                        if (cache && cache.length === list.length) {
                            let same = true;
                            for (let i = 0; i < list.length; i++) {
                                if (cache[i] !== list[i]) {
                                    same = false;
                                    break;
                                }
                            }
                            if (same) return cache;
                        }

                        __listCache[cacheKey] = list;
                        return list;
                    },

            selectTotal: (key) => (s) => s.totals[key] ?? 0,

            //extend hook
            ...(extend ? extend(setTracked, get) : {}),
        };
    });

const employeeMutationStore = createMutationStore((set, get) => ({
  createEmployee: async (dto) => {
    const mutationKey = `employee:create`;
    const { start, success, fail } = get();
    if (!start(mutationKey)) return;

    const listKey = storeKey({ keyName: "employees", scope: "list", params: {} });
    const queryStore = employeeQueryStore.getState();
    const prev = queryStore.queries[listKey] ?? [];

    try {
      const created = await employeeService.createEmployee(dto);
      queryStore.invalidate(listKey);
      success(mutationKey);
      return created;
    } catch (err) {
      fail(mutationKey, err);
      throw err;
    }
  },

  updateEmployee: async (id, dto) => {
    const mutationKey = `employee:update:${id}`;
    const { start, success, fail } = get();
    if (!start(mutationKey)) return;

    const queryStore = employeeQueryStore.getState();
    const prev = queryStore.entities[id];

    try {
      queryStore.setEntity(id, { ...prev, ...dto, __optimistic: true });
      const updated = await employeeService.updateEmployee(id, dto);
      queryStore.setEntity(id, updated);
      success(mutationKey);
      return updated;
    } catch (err) {
      if (prev) queryStore.setEntity(id, prev);
      fail(mutationKey, err);
      throw err;
    }
  },

  activateEmployee: async (id) => {
    const mutationKey = `employee:activate:${id}`;
    const { start, success, fail } = get();
    if (!start(mutationKey)) return;
    try {
      await employeeService.activateEmployee(id);
      const listKey = storeKey({ keyName: "employees", scope: "list", params: {} });
      const queryStore = employeeQueryStore.getState();
      queryStore.invalidate(listKey);
      success(mutationKey);
    } catch (err) {
      fail(mutationKey, err);
      throw err;
    }
